cmake_minimum_required(VERSION 3.20)
project(nmtbench CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
if(APPLE)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(${GGML_DIR} ggml)

set(NMT ${NMT_SRC})
file(GLOB NMT_CPP ${NMT}/nmt*.cpp)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SPM REQUIRED IMPORTED_TARGET sentencepiece)
find_package(absl QUIET)

foreach(t nmtbench2 cntbench)
  add_executable(${t} ${t}.cpp ${NMT_CPP} ${NMT}/NmtLazyInitializeBackend.cpp)
  target_include_directories(${t} PRIVATE stub ${NMT} ${CMAKE_SOURCE_DIR}/addon-cpp)
  target_compile_options(${t} PRIVATE -include ${CMAKE_SOURCE_DIR}/stub/spm_compat.h)
  target_compile_definitions(${t} PRIVATE QVAC_TRANSLATION_NMTCPP_TESTING)
  target_link_libraries(${t} PRIVATE ggml PkgConfig::SPM)
  if(WIN32)
    target_link_libraries(${t} PRIVATE psapi)
  endif()
  if(absl_FOUND)
    target_link_libraries(${t} PRIVATE absl::status absl::strings absl::log)
  endif()
endforeach()
