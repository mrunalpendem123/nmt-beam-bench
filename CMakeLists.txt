cmake_minimum_required(VERSION 3.20)
project(nmtbench CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
if(APPLE)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(${GGML_DIR} ggml)

set(NMT ${NMT_SRC})
file(GLOB NMT_CPP ${NMT}/nmt*.cpp)
# sentencepiece built from source, the same way on every OS
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece
  URL https://github.com/google/sentencepiece/archive/refs/tags/v0.2.0.tar.gz)
FetchContent_MakeAvailable(sentencepiece)
target_compile_options(sentencepiece-static PRIVATE -include ${CMAKE_SOURCE_DIR}/stub/cstdint_fix.h)

foreach(t nmtbench2 cntbench)
  add_executable(${t} ${t}.cpp ${NMT_CPP} ${NMT}/NmtLazyInitializeBackend.cpp)
  target_include_directories(${t} PRIVATE stub ${NMT} ${CMAKE_SOURCE_DIR}/addon-cpp ${sentencepiece_SOURCE_DIR}/src ${sentencepiece_SOURCE_DIR})
  target_compile_options(${t} PRIVATE -include ${CMAKE_SOURCE_DIR}/stub/spm_compat.h)
  target_compile_definitions(${t} PRIVATE QVAC_TRANSLATION_NMTCPP_TESTING)
  target_link_libraries(${t} PRIVATE ggml sentencepiece-static)
  if(WIN32)
    target_link_libraries(${t} PRIVATE psapi)
  endif()
endforeach()
